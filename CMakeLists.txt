cmake_minimum_required(VERSION 3.20)
project(StickyNotes VERSION 1.0.0 LANGUAGES CXX RC)
if(NOT WIN32)
    message(FATAL_ERROR "StickyNotes benötigt Windows.")
endif()
add_executable(StickyNotesApp WIN32 main.cpp)
set_target_properties(StickyNotesApp PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/intern")
target_compile_features(StickyNotesApp PRIVATE cxx_std_17)
target_compile_definitions(StickyNotesApp PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(StickyNotesApp PRIVATE user32 gdi32 shell32 ole32)
set(PORTABLE_RESOURCES "1 RCDATA \"$<TARGET_FILE:StickyNotesApp>\"\n")
if(MINGW)
    target_link_options(StickyNotesApp PRIVATE -municode -static-libgcc -static-libstdc++)
    get_filename_component(COMPILER_DIR "${CMAKE_CXX_COMPILER}" DIRECTORY)
    set(WINPTHREAD_DLL "${COMPILER_DIR}/libwinpthread-1.dll")
    if(EXISTS "${WINPTHREAD_DLL}")
        add_custom_command(TARGET StickyNotesApp POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${WINPTHREAD_DLL}" "$<TARGET_FILE_DIR:StickyNotesApp>")
        string(APPEND PORTABLE_RESOURCES "2 RCDATA \"${WINPTHREAD_DLL}\"\n")
    else()
        message(FATAL_ERROR "Die MinGW-Laufzeit libwinpthread-1.dll wurde nicht gefunden.")
    endif()
endif()

if(MSVC)
    target_compile_options(StickyNotesApp PRIVATE /utf-8)
endif()
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/portable-template-$<CONFIG>.rc" CONTENT "${PORTABLE_RESOURCES}")
add_custom_command(
    OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/portable-$<CONFIG>.rc"
    COMMAND ${CMAKE_COMMAND} -E copy
        "${CMAKE_CURRENT_BINARY_DIR}/portable-template-$<CONFIG>.rc"
        "${CMAKE_CURRENT_BINARY_DIR}/portable-$<CONFIG>.rc"
    DEPENDS StickyNotesApp "${CMAKE_CURRENT_BINARY_DIR}/portable-template-$<CONFIG>.rc"
    VERBATIM
)
add_executable(StickyNotes WIN32 launcher.cpp version.rc "${CMAKE_CURRENT_BINARY_DIR}/portable-$<CONFIG>.rc")
add_dependencies(StickyNotes StickyNotesApp)
target_compile_features(StickyNotes PRIVATE cxx_std_17)
target_compile_definitions(StickyNotes PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(StickyNotes PRIVATE user32)
if(MINGW)
    target_link_options(StickyNotes PRIVATE -municode -static-libgcc -static-libstdc++)
endif()
if(MSVC)
    target_compile_options(StickyNotes PRIVATE /utf-8)
    set_property(TARGET StickyNotesApp StickyNotes PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
