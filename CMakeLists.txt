cmake_minimum_required(VERSION 3.20)
project(StickyNotes VERSION 1.0.3 LANGUAGES CXX RC)
if(NOT WIN32)
    message(FATAL_ERROR "StickyNotes benötigt Windows.")
endif()
set_source_files_properties(appicon.rc PROPERTIES OBJECT_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/assets/stickynotes.ico")
add_executable(StickyNotes WIN32 main.cpp jumplist.cpp version.rc appicon.rc)
target_compile_features(StickyNotes PRIVATE cxx_std_17)
target_compile_definitions(StickyNotes PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(StickyNotes PRIVATE user32 gdi32 shell32 ole32 uuid)
if(MINGW)
    if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
        message(FATAL_ERROR "Die portable MinGW-Version benötigt eine 64-Bit-Toolchain.")
    endif()
    target_link_options(StickyNotes PRIVATE -municode -static -static-libgcc -static-libstdc++)
    target_link_directories(StickyNotes BEFORE PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/third_party/winpthreads")
    set(ERRNO_IMPORT "${CMAKE_CURRENT_BINARY_DIR}/libmsvcrt-errno.a")
    add_custom_command(OUTPUT "${ERRNO_IMPORT}"
        COMMAND "${CMAKE_DLLTOOL}" -d "${CMAKE_CURRENT_SOURCE_DIR}/third_party/winpthreads/msvcrt-errno.def" -l "${ERRNO_IMPORT}" -m i386:x86-64
        DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/third_party/winpthreads/msvcrt-errno.def" VERBATIM)
    add_custom_target(StickyNotesErrnoImport DEPENDS "${ERRNO_IMPORT}")
    add_dependencies(StickyNotes StickyNotesErrnoImport)
    target_link_options(StickyNotes PRIVATE -Wl,-u,__imp__set_errno)
    target_link_libraries(StickyNotes PRIVATE "${ERRNO_IMPORT}")
    get_filename_component(COMPILER_DIR "${CMAKE_CXX_COMPILER}" DIRECTORY)
    set(WINPTHREAD_DLL "${COMPILER_DIR}/libwinpthread-1.dll")
    if(NOT EXISTS "${WINPTHREAD_DLL}")
        message(FATAL_ERROR "Die MinGW-Laufzeit libwinpthread-1.dll wurde nicht gefunden.")
    endif()
endif()

if(MSVC)
    target_compile_options(StickyNotes PRIVATE /utf-8)
    set_property(TARGET StickyNotes PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(StickyNotesReliability EXCLUDE_FROM_ALL tests/reliability.cpp jumplist.cpp)
    set_target_properties(StickyNotesReliability PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/intern")
    target_compile_features(StickyNotesReliability PRIVATE cxx_std_17)
    target_compile_definitions(StickyNotesReliability PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
    target_link_libraries(StickyNotesReliability PRIVATE user32 gdi32 shell32 ole32 uuid)
    if(MINGW)
        target_link_options(StickyNotesReliability PRIVATE -static-libgcc -static-libstdc++)
        add_custom_command(TARGET StickyNotesReliability POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${WINPTHREAD_DLL}" "$<TARGET_FILE_DIR:StickyNotesReliability>")
    endif()
    if(MSVC)
        target_compile_options(StickyNotesReliability PRIVATE /utf-8)
        set_property(TARGET StickyNotesReliability PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    endif()
    add_test(NAME reliability COMMAND StickyNotesReliability)
    add_executable(StickyNotesAttachments EXCLUDE_FROM_ALL tests/attachments.cpp jumplist.cpp)
    set_target_properties(StickyNotesAttachments PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/intern")
    target_compile_features(StickyNotesAttachments PRIVATE cxx_std_17)
    target_compile_definitions(StickyNotesAttachments PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
    target_link_libraries(StickyNotesAttachments PRIVATE user32 gdi32 shell32 ole32 uuid)
    if(MINGW)
        target_link_options(StickyNotesAttachments PRIVATE -static-libgcc -static-libstdc++)
        add_custom_command(TARGET StickyNotesAttachments POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${WINPTHREAD_DLL}" "$<TARGET_FILE_DIR:StickyNotesAttachments>")
    endif()
    if(MSVC)
        target_compile_options(StickyNotesAttachments PRIVATE /utf-8)
        set_property(TARGET StickyNotesAttachments PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    endif()
    add_test(NAME attachments COMMAND StickyNotesAttachments)
endif()
